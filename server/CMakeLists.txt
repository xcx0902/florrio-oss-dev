include(sources.cmake)
add_library(server_runtime STATIC ${SERVER_RUNTIME_SOURCES})
target_include_directories(server_runtime PUBLIC include)
target_link_libraries(server_runtime PUBLIC engine_core florr_boost_headers Threads::Threads)
target_compile_definitions(server_runtime PRIVATE FLORR_SERVER_VERSION="${CMAKE_PROJECT_VERSION}")
  file(GLOB_RECURSE native_identity_files CONFIGURE_DEPENDS
    "${PROJECT_SOURCE_DIR}/server/src/*" "${PROJECT_SOURCE_DIR}/server/include/*"
    "${PROJECT_SOURCE_DIR}/server/config/*" "${PROJECT_SOURCE_DIR}/engine/src/*"
    "${PROJECT_SOURCE_DIR}/engine/include/*" "${PROJECT_SOURCE_DIR}/cmake/*" "${PROJECT_SOURCE_DIR}/schema/*")
  list(APPEND native_identity_files "${PROJECT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/sources.cmake" "${PROJECT_SOURCE_DIR}/engine/sources.cmake" "${PROJECT_SOURCE_DIR}/engine/CMakeLists.txt")
  list(SORT native_identity_files)
  set(native_identity "")
  foreach(file IN LISTS native_identity_files)
    file(SHA256 "${file}" digest)
    file(RELATIVE_PATH relative "${PROJECT_SOURCE_DIR}" "${file}")
    string(APPEND native_identity "${relative}:${digest}\n")
  endforeach()
  string(SHA256 FLORR_NATIVE_SOURCE_ID "${native_identity}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${native_identity_files})
target_compile_definitions(server_runtime PRIVATE FLORR_NATIVE_SOURCE_ID="${FLORR_NATIVE_SOURCE_ID}")
set_target_properties(server_runtime PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_options(server_runtime PRIVATE -Wall -Wextra -Werror)
